```python
class BankAccount:
    def __init__(self, owner, balance=0):
        self.owner = owner
        self._balance = balance

    @property
    def balance(self):
        return self._balance

    def deposit(self, amount):
        if amount > 0:
            self._balance += amount
        else:
            print("Сумма должна быть положительной")

    def withdraw(self, amount):
        if 0 < amount <= self._balance:
            self._balance -= amount
        else:
            print("Недостаточно средств или неверная сумма")

    def __str__(self):
        return f"Владелец: {self.owner}, остаток: {self._balance} тг"

acc.deposit(500)
acc.withdraw(2000)
```

```python
class Student:
    def __init__(self, name):
        self.name = name
        self._grades = []

    def add_grade(self, g):
        if 1 <= g <= 10:
            self._grades.append(g)
        else:
            print("Оценка должна быть в диапазоне 1–10")

    @property
    def average(self):
        if not self._grades:
            return 0
        return round(sum(self._grades) / len(self._grades), 2)

    def __str__(self):
        return f"Имя: {self.name}, средний балл: {self.average}"

    def __eq__(self, other):
        if not isinstance(other, Student):
            return NotImplemented
        return self.average == other.average


a = Student("Ару")
b = Student("Ерасыл")
for g in (8, 10, 9):
    a.add_grade(g)
for g in (9, 9):
    b.add_grade(g)
print(a, b, a == b, sep="\n")
```

```python
class Employee:
    def __init__(self, name, salary):
        self.name = name
        self.salary = salary

    def total_income(self):

class Manager(Employee):
    def __init__(self, name, salary, bonus):
        super().__init__(name, salary)
        self.bonus = bonus

    def total_income(self):
        return super().total_income() + self.bonus

for e in staff:
```
