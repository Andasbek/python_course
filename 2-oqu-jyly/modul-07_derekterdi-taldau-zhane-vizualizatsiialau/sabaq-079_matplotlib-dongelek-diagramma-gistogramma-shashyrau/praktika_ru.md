**Задание 8**
```python
import pandas as pd


def korrelyatsiya(x, y):
    """Коэффициент корреляции Пирсона."""
    n = len(x)
    x_orta = sum(x) / n
    y_orta = sum(y) / n
    alym = 0
    sx = 0
    sy = 0
    for i in range(n):
        dx = x[i] - x_orta
        dy = y[i] - y_orta
        alym += dx * dy
        sx += dx ** 2
        sy += dy ** 2
    return alym / (sx * sy) ** 0.5


baga = [300, 450, 600, 800, 1000, 1200, 1500, 2000]
satylym = [520, 470, 390, 330, 260, 210, 150, 90]
print(round(korrelyatsiya(baga, satylym), 4))
print(round(pd.Series(baga).corr(pd.Series(satylym)), 4))
```
Оба результата должны совпасть.
