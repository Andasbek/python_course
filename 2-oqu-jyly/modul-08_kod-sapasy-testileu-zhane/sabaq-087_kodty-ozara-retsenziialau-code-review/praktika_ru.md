```python
"""Модуль учёта книг школьной библиотеки."""

import sqlite3


class Kitaphana:
    """Библиотека на основе SQLite."""

    def __init__(self, db_path="kitaphana.db"):
        """Подключается к базе данных и создаёт таблицу."""
        self.conn = sqlite3.connect(db_path)
        self.conn.execute(
            "CREATE TABLE IF NOT EXISTS kitaptar ("
            "atauy TEXT PRIMARY KEY, avtory TEXT, zhyly INTEGER, sany INTEGER)"
        )

    def qosu(self, atauy, avtory, zhyly, sany):
        """Добавляет новую книгу."""
        if sany < 0:
            raise ValueError("Количество книг не может быть отрицательным")
        self.conn.execute(
            "INSERT INTO kitaptar VALUES (?, ?, ?, ?)",
            (atauy, avtory, zhyly, sany),
        )
        self.conn.commit()

    def tabu(self, atauy):
        """Находит книгу по названию; если её нет — None."""
        return self.conn.execute(
            "SELECT atauy, avtory, zhyly, sany FROM kitaptar WHERE atauy = ?",
            (atauy,),
        ).fetchone()

    def beru(self, atauy):
        """Выдаёт книгу ученику. При успехе возвращает True."""
        kitap = self.tabu(atauy)
        if kitap is None or kitap[3] == 0:
            return False
        self.conn.execute(
            "UPDATE kitaptar SET sany = sany - 1 WHERE atauy = ?", (atauy,)
        )
        self.conn.commit()
        return True

    def statistika(self, zhyl_shegi=2000):
        """Возвращает общее количество книг и количество книг до zhyl_shegi."""
        barlygy = self.conn.execute(
            "SELECT COALESCE(SUM(sany), 0) FROM kitaptar"
        ).fetchone()[0]
        eski = self.conn.execute(
            "SELECT COALESCE(SUM(sany), 0) FROM kitaptar WHERE zhyly < ?",
            (zhyl_shegi,),
        ).fetchone()[0]
        return barlygy, eski

    def zhabu(self):
        """Закрывает соединение с базой данных."""
        self.conn.close()


if __name__ == "__main__":
    k = Kitaphana(":memory:")
    k.qosu("Путь Абая", "М. Ауэзов", 1942, 3)
    k.qosu("Кочевники", "И. Есенберлин", 1976, 2)
    k.qosu("Основы Python", "А. Автор", 2023, 5)
    print(k.beru("Путь Абая"), k.beru("Несуществующая книга"))
    print("Всего и до 2000 года:", k.statistika())
    k.zhabu()
```

```python
from collections import Counter


def kesteni_tekseru(kesteh):
    """Проверяет расписание уроков и возвращает список предупреждений."""
    eskertuler = []
    for kun, sabaqtar in kesteh.items():
        if not sabaqtar:
            eskertuler.append(f"{kun}: пустой день")
            continue
        if len(sabaqtar) > 7:
            eskertuler.append(f"{kun}: слишком много уроков ({len(sabaqtar)})")
        for pan, san in Counter(sabaqtar).items():
            if san > 2:
                eskertuler.append(f"{kun}: «{pan}» {san} раза")
    return eskertuler


kesteh = {
    "Пн": ["Алгебра", "Физика", "Алгебра", "Алгебра"],
    "Вт": ["История", "Химия", "Биология", "Английский", "Казахский язык", "Литература", "Физкультура", "Информатика"],
    "Ср": [],
}
for e in kesteni_tekseru(kesteh):
    print(e)
```

```python
import ast
import re
import sys


def tekseru(path):
    """Проверяет файл по простым правилам."""
    with open(path, encoding="utf-8") as f:
        matin = f.read()
    eskertuler = []
    for nomir, zhol in enumerate(matin.splitlines(), start=1):
        if len(zhol) > 79:
            eskertuler.append((nomir, f"строка слишком длинная ({len(zhol)} > 79)"))
        if zhol != zhol.rstrip():
            eskertuler.append((nomir, "лишний пробел в конце строки"))
        if re.search(r"execute\(\s*f[\"']", zhol):
            eskertuler.append((nomir, "SQL через f-строку — угроза инъекции, используйте '?'"))

    agash = ast.parse(matin)
    for tuyin in ast.walk(agash):
        if isinstance(tuyin, ast.FunctionDef):
            if not re.fullmatch(r"_{0,2}[a-z][a-z0-9_]*_{0,2}", tuyin.name):
                eskertuler.append((tuyin.lineno, f"имя функции не в стиле snake_case: {tuyin.name}"))
            if ast.get_docstring(tuyin) is None:
                eskertuler.append((tuyin.lineno, f"нет docstring: {tuyin.name}()"))
        elif isinstance(tuyin, ast.ClassDef):
            if not re.fullmatch(r"[A-Z][A-Za-z0-9]*", tuyin.name):
                eskertuler.append((tuyin.lineno, f"имя класса не в стиле CamelCase: {tuyin.name}"))
            if ast.get_docstring(tuyin) is None:
                eskertuler.append((tuyin.lineno, f"нет docstring: class {tuyin.name}"))

    for nomir, habar in sorted(eskertuler):
        print(f"{path}:{nomir}: {habar}")
    print("Всего:", len(eskertuler))


if __name__ == "__main__":
    tekseru(sys.argv[1] if len(sys.argv) > 1 else __file__)
```
