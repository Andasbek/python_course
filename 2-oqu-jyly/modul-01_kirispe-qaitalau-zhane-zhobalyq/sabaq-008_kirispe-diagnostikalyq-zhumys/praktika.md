def count_words(text):
    """Мәтіндегі әр сөздің кездесу санын сөздік ретінде қайтарады."""
    counts = {}
    for word in text.lower().split():
        counts[word] = counts.get(word, 0) + 1
    return counts


print(count_words("Алма алма өрік"))

**1-нұсқа, 4-тапсырма**
```python
EXCELLENT_THRESHOLD = 4.5


def average(grades):
    """Бағалардың орташасын қайтарады."""
    return sum(grades) / len(grades)


def print_students(journal):
    """Әр оқушының орташасын шығарады, үздіктерді белгілейді."""
    for name, grades in journal.items():
        avg = average(grades)
        mark = " (үздік)" if avg > EXCELLENT_THRESHOLD else ""
        print(f"{name}: {avg:.2f}{mark}")


def class_average(journal):
    """Сыныптағы барлық бағалардың орташасын қайтарады."""
    all_grades = [g for grades in journal.values() for g in grades]
    return average(all_grades)


def main():
    journal = {"Әлия": [5, 4, 5], "Бекзат": [3, 4, 3], "Дана": [5, 5, 4], "Ерлан": [4, 3, 4]}
    print_students(journal)
    print(f"Сынып бойынша: {class_average(journal):.2f}")


if __name__ == "__main__":
    main()
```

    """Қойманың жалпы құнын қайтарады."""
    return sum(item_value(count, price) for count, price in store.values())

def read_journal(path):
    """Файлдан журналды сөздік ретінде оқиды."""
    journal = {}
    with open(path, encoding="utf-8") as file:
        for line in file:
            line = line.strip()
            if not line:
                continue
            name, grades_text = line.split(";")
            journal[name] = [int(g) for g in grades_text.split(",")]
    return journal


def write_report(journal, path):
    """Әр оқушының орташасын файлға жазады."""
    with open(path, "w", encoding="utf-8") as file:
        for name, grades in journal.items():
            file.write(f"{name}: {sum(grades) / len(grades):.2f}\n")


if __name__ == "__main__":
    write_report(read_journal("grades.txt"), "report.txt")
