import json
from collections import Counter

ULGI = [
    {"id": "US-1", "matin": "Выбрать тему и пройти тест", "basymdyq": "M",
     "kriteriler": ["Показываются 10 вопросов", "Можно выбрать ответ"]},

    {"id": "US-3", "matin": "График по темам", "basymdyq": "M", "kriteriler": []},
    {"id": "US-4", "matin": "Импорт из CSV", "basymdyq": "S", "kriteriler": []},
    {"id": "US-5", "matin": "Таймер", "basymdyq": "C", "kriteriler": []},
]
ATAULAR = {"M": "Must", "S": "Should", "C": "Could", "W": "Won't"}


def zhukteu(path="requirements.json"):
    """Читает требования из JSON; если файла нет, сохраняет образец."""
    try:
        with open(path, encoding="utf-8") as f:
            return json.load(f)
    except FileNotFoundError:
        with open(path, "w", encoding="utf-8") as f:
            json.dump(ULGI, f, ensure_ascii=False, indent=2)
        return ULGI


def taldau(talaptar):
    """Статистика приоритетов и предупреждения о Must без критериев."""
    sanau = Counter(t["basymdyq"] for t in talaptar)
    for kod, atau in ATAULAR.items():
        print(f"{atau:<7} {sanau.get(kod, 0)}")
    for t in talaptar:
        if t["basymdyq"] == "M" and not t["kriteriler"]:
            print(f"Предупреждение: {t['id']} (Must) — нет критериев приёмки")


def kesteh_zhasau(talaptar, path="TT_kesteh.md"):
    """Создаёт Markdown-таблицу для отслеживания прогресса."""
    zholdar = ["| ID | Требование | Приоритет | Число критериев | Статус |", "|---|---|---|---|---|"]
    for t in talaptar:
        zholdar.append(f"| {t['id']} | {t['matin']} | {ATAULAR[t['basymdyq']]} "
                       f"| {len(t['kriteriler'])} | [ ] |")
    with open(path, "w", encoding="utf-8") as f:
        f.write("\n".join(zholdar) + "\n")
    print(f"{path} создан")


talaptar = zhukteu()
taldau(talaptar)
kesteh_zhasau(talaptar)
