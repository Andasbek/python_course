**Задание 8.**
```python
"""tools/make_qr.py — QR-код для ссылки на репозиторий."""
import sys

import qrcode


def make_qr(url, filename="docs/repo_qr.png"):
    qr = qrcode.QRCode(box_size=10, border=2)
    qr.add_data(url)
    qr.make(fit=True)
    image = qr.make_image(fill_color="black", back_color="white")
    image.save(filename)
    return filename


if __name__ == "__main__":
    link = sys.argv[1] if len(sys.argv) > 1 else "https://github.com/<логин>/qalta-aqshasy"
    print("Сохранено:", make_qr(link))
```
Запуск: `python tools/make_qr.py https://github.com/<логин>/<репозиторий>`.
