```python
import tkinter as tk
from tkinter import ttk, messagebox

root = tk.Tk()
root.title("Карманные деньги — добавление расхода")
frame = ttk.Frame(root, padding=10)
frame.pack()

ttk.Label(frame, text="Сумма (тг):").grid(row=0, column=0, sticky="w", pady=4)
amount_entry = ttk.Entry(frame, width=20)
amount_entry.grid(row=0, column=1, pady=4)

ttk.Label(frame, text="Категория:").grid(row=1, column=0, sticky="w", pady=4)
category_box = ttk.Combobox(frame, values=["Еда", "Транспорт", "Развлечения"],
                            state="readonly", width=18)
category_box.current(0)
category_box.grid(row=1, column=1, pady=4)

status = ttk.Label(frame, text="")
status.grid(row=3, column=0, columnspan=2)


def save():
    text = amount_entry.get().strip()
    if not text.isdigit() or int(text) == 0:
        messagebox.showwarning("Ошибка", "Введите сумму в виде целого положительного числа.")
        return
    status.config(text=f"Сохранено: {text} тг, {category_box.get()}")
    amount_entry.delete(0, "end")


def delete_all():
    if messagebox.askyesno("Подтверждение", "Удалить все операции? Это действие нельзя отменить."):
        status.config(text="Всё удалено")


ttk.Button(frame, text="Сохранить", command=save).grid(row=2, column=1, sticky="e", pady=6)
tk.Button(frame, text="Удалить всё", command=delete_all,
          bg="#c62828", fg="white").grid(row=2, column=0, sticky="w", pady=6)
root.mainloop()
```

**Задание 7.**
```python
import tkinter as tk
from tkinter import ttk

root = tk.Tk()
root.title("Прототип экранов")
root.geometry("400x250")
root.rowconfigure(0, weight=1)
root.columnconfigure(0, weight=1)

screens = {}


def show(name):
    screens[name].tkraise()


for name in ("Главный", "История", "Статистика"):
    frame = ttk.Frame(root, padding=20)
    frame.grid(row=0, column=0, sticky="nsew")
    ttk.Label(frame, text=f"Экран «{name}»", font=("Arial", 16)).pack(pady=10)
    screens[name] = frame

for name in ("История", "Статистика"):
    ttk.Button(screens["Главный"], text=name,
               command=lambda n=name: show(n)).pack(pady=4)
    ttk.Button(screens[name], text="← Назад",
               command=lambda: show("Главный")).pack(pady=4)

show("Главный")
root.mainloop()
```
Обратите внимание: `lambda n=name: show(n)` — параметр по умолчанию используется, чтобы «зафиксировать» переменную цикла; без него все кнопки открывали бы последний экран.

```python
import tkinter as tk

THEMES = {
    "light": {"bg": "#ffffff", "fg": "#222222", "btn": "#e0e0e0"},
    "dark": {"bg": "#202124", "fg": "#f1f1f1", "btn": "#3c4043"},
}
current = "light"

root = tk.Tk()
root.title("Переключатель темы")
label = tk.Label(root, text="Баланс: 12 450 тг", font=("Arial", 16))
label.pack(padx=20, pady=20)
button = tk.Button(root, text="Ночной режим")
button.pack(pady=10)


def apply_theme():
    colors = THEMES[current]
    root.config(bg=colors["bg"])
    label.config(bg=colors["bg"], fg=colors["fg"])
    button.config(bg=colors["btn"], fg=colors["fg"],
                  text="Дневной режим" if current == "dark" else "Ночной режим")


def toggle():
    global current
    current = "dark" if current == "light" else "light"
    apply_theme()


button.config(command=toggle)
apply_theme()
root.mainloop()
```
