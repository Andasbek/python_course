**Задание 7**
```python
import json

import matplotlib.pyplot as plt

SALMAQTAR = {
    "ozektilik": 2,
    "derbestik": 1,
    "zhuzege_asyru": 2,
    "tehnologiyalar": 1,
    "qyzyqtylyq": 2,
}

ULGI = {
    "Тренажёр ЕНТ": {"ozektilik": 5, "derbestik": 3, "zhuzege_asyru": 5,
                     "tehnologiyalar": 5, "qyzyqtylyq": 4},
    "Монитор воздуха": {"ozektilik": 5, "derbestik": 4, "zhuzege_asyru": 4,
                        "tehnologiyalar": 5, "qyzyqtylyq": 5},
    "3D-шутер": {"ozektilik": 2, "derbestik": 2, "zhuzege_asyru": 1,
                 "tehnologiyalar": 2, "qyzyqtylyq": 5},
}


def zhukteu(path="ideas.json"):
    """Читает оценки идей из JSON; если файла нет, сохраняет образец и возвращает его."""
    try:
        with open(path, encoding="utf-8") as f:
            return json.load(f)
    except FileNotFoundError:
        with open(path, "w", encoding="utf-8") as f:
            json.dump(ULGI, f, ensure_ascii=False, indent=2)
        return ULGI


def esepteu(ideyalar):
    """Считает взвешенный балл каждой идеи и возвращает их по убыванию."""
    natije = {}
    for atau, bagalar in ideyalar.items():
        natije[atau] = sum(bagalar[k] * s for k, s in SALMAQTAR.items())
    return sorted(natije.items(), key=lambda x: x[1], reverse=True)


reiting = esepteu(zhukteu())
for orny, (atau, upai) in enumerate(reiting, start=1):
    print(f"{orny}. {atau:<22} {upai}/40")

ataular = [r[0] for r in reiting][::-1]
upailar = [r[1] for r in reiting][::-1]
tuster = ["#2A9D8F" if u == max(upailar) else "lightgray" for u in upailar]
fig, ax = plt.subplots(figsize=(7, 3.5))
b = ax.barh(ataular, upailar, color=tuster)
ax.bar_label(b)
ax.set_xlim(0, 40)
ax.set_xlabel("Взвешенный балл (макс. 40)")
ax.set_title("Рейтинг идей", loc="left", fontweight="bold")
ax.spines[["top", "right"]].set_visible(False)
plt.tight_layout()
plt.show()
```
Результат: `1. Монитор воздуха 37/40`, `2. Тренажёр ЕНТ 36/40`, `3. 3D-шутер 20/40`.
