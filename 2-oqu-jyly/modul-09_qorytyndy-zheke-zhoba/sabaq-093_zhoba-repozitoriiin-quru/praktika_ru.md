**Задание 7.**
```python
import csv
import datetime
import sys


def load(path):
    with open(path, encoding="utf-8") as f:
        return list(csv.DictReader(f))


def parse_date(text):
    """'26.04.27' -> datetime.date(2027, 4, 26)"""
    return datetime.datetime.strptime(text, "%d.%m.%y").date()


def report(rows, today):
    done = sum(1 for r in rows if r["status"] == "done")
    percent = done * 100 // len(rows) if rows else 0
    bar = "#" * (percent // 10) + "-" * (10 - percent // 10)
    print(f"Прогресс: {percent}% [{bar}]")
    for r in rows:
        if r["status"] != "done" and parse_date(r["deadline"]) < today:
            print(f"Просрочено: {r['task']} ({r['deadline']})")


if __name__ == "__main__":
    path = sys.argv[1] if len(sys.argv) > 1 else "docs/plan.csv"
    report(load(path), datetime.date.today())
```

**Задание 8.** Поскольку задачи даны в порядке зависимостей, «самое раннее время завершения» можно вычислить за один проход (в общем случае нужна топологическая сортировка).
```python
tasks = {
    "A": (2, []),
    "B": (3, ["A"]),
    "C": (2, ["A"]),
    "D": (4, ["B", "C"]),
    "E": (1, ["B"]),
    "F": (2, ["D", "E"]),
}

finish = {}   # самое раннее время завершения задачи
prev = {}     # предыдущая задача на критическом пути


def earliest_finish(name):
    if name in finish:
        return finish[name]
    duration, deps = tasks[name]
    start = 0
    prev[name] = None
    for d in deps:
        if earliest_finish(d) > start:
            start = earliest_finish(d)
            prev[name] = d
    finish[name] = start + duration
    return finish[name]


last = max(tasks, key=earliest_finish)
path = []
node = last
while node is not None:
    path.append(node)
    node = prev[node]
print(f"Минимальное время: {finish[last]} ч, путь: {' -> '.join(reversed(path))}")
```
Результат:
```text
Минимальное время: 11 ч, путь: A -> B -> D -> F
```
