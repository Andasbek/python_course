**Задание 5.** Ошибки: (1) вместо `>` должно быть `>=` — при ровно 85% выходит 4, а не 5 (то же для 65% и 40%); (2) `max_score = 0` → `ZeroDivisionError`; (3) не проверяются `score > max_score` и отрицательные баллы.
```python
import unittest


def grade(score, max_score):
    """Оценка по проценту: 85%+ -> 5, 65%+ -> 4, 40%+ -> 3, иначе 2."""
    if max_score <= 0:
        raise ValueError("Максимальный балл должен быть положительным")
    if not 0 <= score <= max_score:
        raise ValueError("Балл должен быть в диапазоне от 0 до максимального")
    percent = score / max_score * 100
    if percent >= 85:
        return 5
    if percent >= 65:
        return 4
    if percent >= 40:
        return 3
    return 2


class GradeTest(unittest.TestCase):
    def test_boundaries(self):
        self.assertEqual(grade(85, 100), 5)
        self.assertEqual(grade(84, 100), 4)
        self.assertEqual(grade(65, 100), 4)
        self.assertEqual(grade(40, 100), 3)
        self.assertEqual(grade(39, 100), 2)

    def test_zero_max(self):
        with self.assertRaises(ValueError):
            grade(5, 0)

    def test_out_of_range(self):
        for score in (-1, 101):
            with self.subTest(score=score):
                with self.assertRaises(ValueError):
                    grade(score, 100)


if __name__ == "__main__":
    unittest.main()
```
Примечание: вычисления с дробными числами неточны: `11 / 20 * 100` = `55.00000000000001`, а `0.57 * 100` = `56.99999999999999`. На границе такой «хвост» может изменить оценку. Более надёжный вариант — сравнение без дробей: `if score * 100 >= 85 * max_score`.

**Задание 9.**
```python
import random
import unittest

MAX_AMOUNT = 10_000_000


def parse_amount(text):
    cleaned = text.strip().replace(" ", "")
    if not cleaned.isdigit():
        raise ValueError("Введите сумму числом")
    value = int(cleaned)
    if value <= 0:
        raise ValueError("Сумма должна быть положительной")
    if value > MAX_AMOUNT:
        raise ValueError("Сумма слишком большая")
    return value


ALPHABET = "0123456789 -.,абвәғқңөүұһіAZ²٣"


class FuzzTest(unittest.TestCase):
    def test_random_strings(self):
        rnd = random.Random(42)
        for _ in range(10_000):
            text = "".join(rnd.choice(ALPHABET) for _ in range(rnd.randint(0, 12)))
            try:
                result = parse_amount(text)
            except ValueError:
                continue
            self.assertIsInstance(result, int, msg=repr(text))
            self.assertTrue(1 <= result <= MAX_AMOUNT, msg=repr(text))


if __name__ == "__main__":
    unittest.main()
```
Интересный результат: `"²".isdigit()` → `True`, но `int("²")` → `ValueError`; а для арабской цифры `"٣"` `int("٣")` → `3`. То есть проверка `isdigit()` не означает «только цифры 0–9». Fuzz-тест находит такие неожиданные случаи; более строгая проверка: `cleaned.isascii() and cleaned.isdigit()`.
