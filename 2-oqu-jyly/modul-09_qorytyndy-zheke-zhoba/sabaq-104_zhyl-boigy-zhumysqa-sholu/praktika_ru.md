**Задание 8.**
```python
"""tools/portfolio_audit.py — проверка портфолио GitHub (открытый API без ключа)."""
import sys

import requests

API = "https://api.github.com"
HEADERS = {"Accept": "application/vnd.github+json"}


def get_repos(login):
    response = requests.get(f"{API}/users/{login}/repos",
                            params={"per_page": 100, "sort": "updated"},
                            headers=HEADERS, timeout=10)
    response.raise_for_status()
    return [r for r in response.json() if not r["fork"]]


def has_readme(login, repo):
    response = requests.get(f"{API}/repos/{login}/{repo}/readme", headers=HEADERS, timeout=10)
    return response.status_code == 200


def audit(login):
    repos = get_repos(login)
    if not repos:
        print("Открытые репозитории не найдены.")
        return
    points, no_description = 0, 0
    for r in repos:
        readme = has_readme(login, r["name"])
        described = bool(r["description"])
        topics = len(r.get("topics", []))
        points += readme + described + (topics > 0)
        no_description += not described
        print(f"{r['name']:<20} {r['language'] or '-':<7} {r['pushed_at'][:10]}  "
              f"описание: {'да' if described else 'нет':<3} topics: {topics}  "
              f"README: {'да' if readme else 'нет'}")
    score = round(points / (3 * len(repos)) * 10)
    print(f"Баллы портфолио: {score}/10.", end=" ")
    if no_description:
        print(f"Совет: добавьте описание в {no_description} репозитори(й/я/ев).")
    else:
        print("Отлично!")


if __name__ == "__main__":
    try:
        audit(sys.argv[1] if len(sys.argv) > 1 else "octocat")
    except requests.RequestException as err:
        print("Не удалось подключиться к GitHub:", err)
```
Примечание: `True + True` = 2 — в Python тип `bool` является наследником `int` (наследование, урок 15), поэтому логические значения можно складывать — интересная тема для обсуждения.

```python
import json

PORTFOLIO = json.loads("""[
  {"title": "Библиотека", "year": 2026, "module": "ООП", "tech": ["ООП"],
   "url": "https://github.com/example/library"},
  {"title": "Карманные деньги", "year": 2027, "module": "Итоговый проект",
   "tech": ["Tkinter", "SQLite", "Matplotlib", "API"],
   "url": "https://github.com/example/qalta-aqshasy"}
]""")


def projects_table(items):
    lines = ["| Проект | Год | Модуль | Технологии |", "|---|---|---|---|"]
    for p in sorted(items, key=lambda p: p["year"], reverse=True):
        lines.append(f"| [{p['title']}]({p['url']}) | {p['year']} | {p['module']} | "
                     f"{', '.join(p['tech'])} |")
    return "\n".join(lines)


print(projects_table(PORTFOLIO))
```
