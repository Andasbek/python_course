def add_record(table, record, key="id", unique=("email",)):
    """Добавляет запись, проверяя ограничения PRIMARY KEY и UNIQUE."""
    if record.get(key) is None:
        print(f"Ошибка: {key} не должен быть пустым")
        return False
    for row in table:
        if row[key] == record[key]:
            print(f"Ошибка: {key}={record[key]} уже есть (повтор PRIMARY KEY)")
            return False
        for field in unique:
            if field in record and row.get(field) == record[field]:
                print(f"Ошибка: {field}={record[field]} уже есть (UNIQUE)")
                return False
    table.append(record)
    return True

classes = [(1, "9А", "Ермекова Г.С."), (2, "9Б", "Оспанова А.Б."), (3, "10А", "Касымов Т.Н.")]
students = [(1, "Айгерим", 1), (2, "Бауыржан", 1), (3, "Даулет", 2), (4, "Сымбат", 9)]


def check_integrity(students, classes):
    ids = {c[0] for c in classes}
    return [s for s in students if s[2] not in ids]


def delete_class(class_id):
    if any(s[2] == class_id for s in students):
        print(f"Класс {class_id} не удаляется: в нём есть ученики")
        return
    classes[:] = [c for c in classes if c[0] != class_id]
    print(f"Класс {class_id} удалён")


print("Нарушена целостность:", check_integrity(students, classes))
delete_class(1)
delete_class(3)
print("Оставшиеся классы:", [c[1] for c in classes])

**Задание 8**

```python
import random
import time

random.seed(1)
table = [{"id": i, "iin": f"{random.randint(0, 10**12 - 1):012d}", "name": f"Гражданин {i}"}
         for i in range(1, 100_001)]
queries = [random.choice(table)["iin"] for _ in range(1000)]

start = time.perf_counter()
for q in queries:
    for row in table:              # полный просмотр
        if row["iin"] == q:
            break
scan_time = time.perf_counter() - start

start = time.perf_counter()
index = {row["iin"]: row for row in table}   # построение индекса
build_time = time.perf_counter() - start
start = time.perf_counter()
for q in queries:
    row = index[q]
index_time = time.perf_counter() - start

print(f"Полный просмотр: {scan_time:.3f} с")
print(f"Построение индекса: {build_time:.3f} с, поиск по индексу: {index_time:.5f} с")
print(f"Поиск по индексу быстрее примерно в {scan_time / index_time:.0f} раз")
```

Результат зависит от компьютера, но поиск по индексу оказывается в тысячи раз быстрее. Объяснение: база данных ведёт «каталог» (B-дерево) по первичному ключу и индексированным полям, поэтому находит запись, не просматривая всю таблицу. Цена индекса — дополнительная память и время на обновление индекса при добавлении/изменении записи.
