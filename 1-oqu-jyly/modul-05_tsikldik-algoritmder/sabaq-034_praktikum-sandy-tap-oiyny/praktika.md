**Практикум — толық үлгі (қиындық деңгейі мен рекордпен)**
```python
import random

print("=== «САНДЫ ТАП» ОЙЫНЫ ===")
record = 0
total_score = 0
play = "иә"
while play == "иә":
    level = input("Деңгей (1 - оңай, 2 - орташа, 3 - қиын): ")
    if level == "1":
        top = 10
        max_attempts = 5
    elif level == "3":
        top = 1000
        max_attempts = 10
    else:
        top = 100
        max_attempts = 7
    secret = random.randint(1, top)
    attempts = 0
    won = False
    print(f"1-ден {top}-ге дейін сан ойладым. Әрекет саны: {max_attempts}")
    while attempts < max_attempts:
        guess = int(input(f"{attempts + 1}-әрекет: "))
        if guess < 1 or guess > top:
            print(f"Сан 1 мен {top} аралығында болуы керек!")
            continue
        attempts += 1
        diff = abs(guess - secret)
        if diff == 0:
            won = True
            break
        if guess < secret:
            print("Көбірек!", end=" ")
        else:
            print("Азырақ!", end=" ")
        if diff < 5:
            print("Өте ыстық!")
        elif diff < 15:
            print("Жылы")
        else:
            print("Суық")
    if won:
        score = (max_attempts - attempts + 1) * 10
        total_score += score
        print(f"Таптың! Әрекет: {attempts}, ұпай: {score}")
        if record == 0 or attempts < record:
            record = attempts
            print("Жаңа рекорд!")
    else:
        print("Әрекет таусылды. Ойлаған саным:", secret)
    play = input("Тағы ойнайсың ба? (иә/жоқ): ")
print("Жалпы ұпай:", total_score, "Рекорд:", record)
```

**★ Жартас-Қайшы-Қағаз**
```python
import random

player_wins = 0
comp_wins = 0
while player_wins < 3 and comp_wins < 3:
    p = int(input("1 - Жартас, 2 - Қайшы, 3 - Қағаз: "))
    if p < 1 or p > 3:
        print("Қате таңдау!")
        continue
    c = random.randint(1, 3)
    print("Компьютер таңдады:", c)
    if p == c:
        print("Тең!")
    elif (p == 1 and c == 2) or (p == 2 and c == 3) or (p == 3 and c == 1):
        print("Сен жеңдің!")
        player_wins += 1
    else:
        print("Компьютер жеңді!")
        comp_wins += 1
    print(f"Есеп: {player_wins} : {comp_wins}")
if player_wins == 3:
    print("Ойынды сен жеңдің!")
else:
    print("Ойынды компьютер жеңді!")
```
