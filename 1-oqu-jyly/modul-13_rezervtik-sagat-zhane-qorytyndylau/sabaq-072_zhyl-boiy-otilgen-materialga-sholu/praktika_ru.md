**Задание 8:**

```python
class Student:
    def __init__(self, name, grades):
        self.name = name
        self.grades = grades

    def average(self):
        return sum(self.grades) / len(self.grades)

    def add_grade(self, grade):
        if 1 <= grade <= 5:
            self.grades.append(grade)

    def is_excellent(self):
        return self.average() >= 4.5


s = Student("Дана", [5, 4])
s.add_grade(5)
print(s.name, round(s.average(), 2), s.is_excellent())
```
