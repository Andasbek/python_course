## Кеңейтілген деңгей

### 1-тапсырма. Жинақты кеңейту

### 2-тапсырма. Қатеге төзімділік
Әр функция қате мәліметке «ақылды» жауап берсін: теріс санның факториалы — `None`, бос тізімнің орташасы — `0`, үшбұрыш жоқ болса — `0` және т.б. Осы жағдайларға да `assert` жазыңыз.

**Практикум, A тақырыбы «Математика көмекшісі» (үлгі)**
```python
"""Математика көмекшісі — функциялар жинағы."""


def is_even(n):
    """n жұп болса True қайтарады."""
    return n % 2 == 0


def is_prime(n):
    """n жай сан болса True қайтарады."""
    if n < 2:
        return False
    for d in range(2, int(n ** 0.5) + 1):
        if n % d == 0:
            return False
    return True


def factorial(n):
    """n! мәнін қайтарады (рекурсивті); теріс n үшін None."""
    if n < 0:
        return None
    if n <= 1:
        return 1
    return n * factorial(n - 1)


def gcd(a, b):
    """Евклид алгоритмімен ЕҮОБ қайтарады."""
    while b != 0:
        a, b = b, a % b
    return a


def lcm(a, b):
    """ЕКОЕ қайтарады (gcd функциясын қолданады)."""
    return a * b // gcd(a, b)


def digit_sum(n):
    """Санның цифрлар қосындысын қайтарады."""
    total = 0
    for ch in str(abs(n)):
        total += int(ch)
    return total


def power(a, n=2):
    """a санының n дәрежесін қайтарады (әдепкі — квадрат)."""
    result = 1
    for i in range(n):
        result *= a
    return result


# ===== Тесттер =====
assert is_even(4) and not is_even(7)
assert is_prime(13) and not is_prime(1) and not is_prime(15)
assert factorial(5) == 120 and factorial(0) == 1 and factorial(-1) is None
assert gcd(12, 18) == 6 and lcm(4, 6) == 12
assert digit_sum(2027) == 11
assert power(3) == 9 and power(2, 10) == 1024
print("Барлық тест өтті!")


# ===== Мәзір =====
def show_menu():
    print("1-жұп па, 2-жай ма, 3-факториал, 4-ЕҮОБ/ЕКОЕ, 5-цифрлар қосындысы, 6-дәреже, 0-шығу")


while True:
    show_menu()
    c = input("Таңдау: ")
    if c == "0":
        break
    elif c == "1":
        n = int(input("n = "))
        print("Жұп" if is_even(n) else "Тақ")
    elif c == "2":
        n = int(input("n = "))
        print("Жай сан" if is_prime(n) else "Жай сан емес")
    elif c == "3":
        n = int(input("n = "))
        print(f"{n}! = {factorial(n)}")
    elif c == "4":
        a = int(input("a = "))
        b = int(input("b = "))
        print("ЕҮОБ:", gcd(a, b), "ЕКОЕ:", lcm(a, b))
    elif c == "5":
        print(digit_sum(int(input("n = "))))
    elif c == "6":
        a = int(input("a = "))
        n = int(input("n = "))
        print(power(a, n))
    else:
        print("Қате таңдау")
```
Ескерту: `"Жұп" if ... else "Тақ"` — шартты өрнек (бір жолдық if). Оқушыларға таныс емес болса, кәдімгі `if/else` түрінде жазуды ұсыныңыз.

def quiz(questions):
    """Сұрақтар сөздігі бойынша викторина өткізіп, ұпайды қайтарады."""
    score = 0
    for q, answer in questions.items():
        if input(q + " ").strip().lower() == answer.lower():
            score += 1
    return score


def show_score(score, total):
    print(f"Нәтиже: {score}/{total}")


questions = {"Қазақстанның астанасы?": "Астана", "5 * 6 = ?": "30"}
total_score = 0
s = quiz(questions)
show_score(s, len(questions))
total_score += s
```

def rectangle(a, b):
    return a * b, 2 * (a + b)


def greet(name, lang="kz"):
    if lang == "ru":
        return f"Привет, {name}!"
    if lang == "en":
        return f"Hello, {name}!"
    return f"Сәлем, {name}!"


def count_positive(numbers):
    count = 0
    for n in numbers:
        if n > 0:
            count += 1
    return count


def sum_digits(n):
    if n < 10:
        return n
    return n % 10 + sum_digits(n // 10)


area, perimeter = rectangle(5, 3)
print(f"Аудан: {area}, периметр: {perimeter}")
print(greet("Дана"))
print(greet("Дана", "ru"))
print(greet("Дана", lang="en"))
print(count_positive([3, -1, 0, 7, -5, 2]))
print(sum_digits(4096))
```

def circle(r):
    return 3.14 * r * r, 2 * 3.14 * r


def price_after(price, discount=10):
    return price - price * discount / 100


def longest_word(text):
    best = ""
    for w in text.split():
        if len(w) > len(best):
            best = w
    return best


def power(a, n):
    if n == 0:
        return 1
    return a * power(a, n - 1)


area, length = circle(10)
print(f"Аудан: {area:.1f}, ұзындық: {length:.1f}")
print(price_after(5000))
