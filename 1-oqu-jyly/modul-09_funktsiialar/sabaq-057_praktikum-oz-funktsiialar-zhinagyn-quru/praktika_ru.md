```python
"""Помощник по математике — сборник функций."""


def is_even(n):
    """Возвращает True, если n чётное."""
    return n % 2 == 0


def is_prime(n):
    """Возвращает True, если n — простое число."""
    if n < 2:
        return False
    for d in range(2, int(n ** 0.5) + 1):
        if n % d == 0:
            return False
    return True


def factorial(n):
    """Возвращает значение n! (рекурсивно); для отрицательного n — None."""
    if n < 0:
        return None
    if n <= 1:
        return 1
    return n * factorial(n - 1)


def gcd(a, b):
    """Возвращает НОД по алгоритму Евклида."""
    while b != 0:
        a, b = b, a % b
    return a


def lcm(a, b):
    """Возвращает НОК (использует функцию gcd)."""
    return a * b // gcd(a, b)


def digit_sum(n):
    """Возвращает сумму цифр числа."""
    total = 0
    for ch in str(abs(n)):
        total += int(ch)
    return total


def power(a, n=2):
    """Возвращает a в степени n (по умолчанию — квадрат)."""
    result = 1
    for i in range(n):
        result *= a
    return result


# ===== Тесты =====
assert is_even(4) and not is_even(7)
assert is_prime(13) and not is_prime(1) and not is_prime(15)
assert factorial(5) == 120 and factorial(0) == 1 and factorial(-1) is None
assert gcd(12, 18) == 6 and lcm(4, 6) == 12
assert digit_sum(2027) == 11
assert power(3) == 9 and power(2, 10) == 1024
print("Все тесты пройдены!")


# ===== Меню =====
def show_menu():
    print("1-чётное ли, 2-простое ли, 3-факториал, 4-НОД/НОК, 5-сумма цифр, 6-степень, 0-выход")


while True:
    show_menu()
    c = input("Выбор: ")
    if c == "0":
        break
    elif c == "1":
        n = int(input("n = "))
        print("Чётное" if is_even(n) else "Нечётное")
    elif c == "2":
        n = int(input("n = "))
        print("Простое число" if is_prime(n) else "Не простое число")
    elif c == "3":
        n = int(input("n = "))
        print(f"{n}! = {factorial(n)}")
    elif c == "4":
        a = int(input("a = "))
        b = int(input("b = "))
        print("НОД:", gcd(a, b), "НОК:", lcm(a, b))
    elif c == "5":

    elif c == "6":
```

```python
def quiz(questions):
    """Проводит викторину по словарю вопросов и возвращает количество очков."""
    score = 0
    for q, answer in questions.items():
        if input(q + " ").strip().lower() == answer.lower():
            score += 1
    return score


def show_score(score, total):
    print(f"Результат: {score}/{total}")


questions = {"Столица Казахстана?": "Астана", "5 * 6 = ?": "30"}
total_score = 0
s = quiz(questions)
show_score(s, len(questions))
total_score += s
```

```python
    count = 0
    for n in numbers:
        if n > 0:
```
