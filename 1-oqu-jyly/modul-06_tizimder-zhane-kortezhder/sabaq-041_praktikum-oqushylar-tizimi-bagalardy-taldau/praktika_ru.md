**Ввод:** количество учеников `n`, затем имя и оценка каждого ученика (2–5; при ошибке — спросить снова).

**Вывод (отчёт):**

**Проект 1. «Список учеников» — образец решения**
```python
students = [("Айгерим", 5), ("Бекзат", 4), ("Дана", 3)]

while True:
    print("\n===== КЛАССНЫЙ ЖУРНАЛ =====")
    print("1. Добавить ученика")
    print("2. Показать всех учеников")
    print("3. Удалить ученика")
    print("4. Найти ученика")
    print("5. Изменить оценку")
    print("6. Сортировать по алфавиту")
    print("7. Сортировать по оценке")
    print("0. Выход")
    choice = input("Ваш выбор: ")

    if choice == "1":
        name = input("Имя: ")
        grade = int(input("Оценка (2-5): "))
        while grade < 2 or grade > 5:
            grade = int(input("Ошибка! Оценка от 2 до 5: "))
        students.append((name, grade))
        print("Добавлен!")
    elif choice == "2":
        if not students:
            print("Список пуст")
        for num, (name, grade) in enumerate(students, start=1):
            print(f"{num}. {name:<12} {grade}")
    elif choice == "3":
        num = int(input("Номер ученика: "))
        if 1 <= num <= len(students):
            removed = students.pop(num - 1)
            print("Удалён:", removed[0])
        else:
            print("Такого номера нет")
    elif choice == "4":
        target = input("Имя: ")
        found = False
        for name, grade in students:
            if name == target:
                print(f"{name} — оценка {grade}")
                found = True
        if not found:
            print("Не найден")
    elif choice == "5":
        num = int(input("Номер ученика: "))
        if 1 <= num <= len(students):
            new_grade = int(input("Новая оценка: "))
            if 2 <= new_grade <= 5:
                name = students[num - 1][0]
                students[num - 1] = (name, new_grade)
                print("Изменено")
            else:
                print("Неверная оценка")
        else:
            print("Такого номера нет")
    elif choice == "6":
        students.sort()
        print("Отсортировано по алфавиту")
    elif choice == "7":
        students.sort(key=lambda s: s[1], reverse=True)
        print("Отсортировано по оценке")
    elif choice == "0":
        print("До свидания!")
        break
    else:
        print("Неверный выбор")
```

**Проект 2. «Анализ оценок» — образец решения (параллельные списки)**
```python
n = int(input("Количество учеников: "))
names = []
grades = []
for i in range(n):
    names.append(input(f"Имя ученика {i + 1}: "))
    g = int(input("Оценка: "))
    while g < 2 or g > 5:
        g = int(input("Ошибка! 2-5: "))
    grades.append(g)

if n > 0:
    avg = sum(grades) / n
    s = sorted(grades)
    if n % 2 == 1:
        median = s[n // 2]
    else:
        median = (s[n // 2 - 1] + s[n // 2]) / 2
    top = max(grades)
    low = min(grades)
    top_names = [names[i] for i in range(n) if grades[i] == top]
    low_names = [names[i] for i in range(n) if grades[i] == low]

    print("===== АНАЛИЗ ОЦЕНОК =====")
    print("Количество учеников:", n)
    print(f"Средняя оценка: {avg:.2f}")
    print("Медиана:", median)
    print(f"Наивысшая оценка: {top} ({', '.join(top_names)})")
    print(f"Наименьшая оценка: {low} ({', '.join(low_names)})")
    print("Распределение:")
    for mark in range(5, 1, -1):
        c = grades.count(mark)
        print(f"  «{mark}»: {'█' * c} {c}")
    quality = (grades.count(4) + grades.count(5)) / n * 100
    success = (n - grades.count(2)) / n * 100
    print(f"Показатель качества: {quality:.1f}%")
    print(f"Показатель успеваемости: {success:.1f}%")
    above = [names[i] for i in range(n) if grades[i] > avg]
    print("Выше среднего:", above)
    for i in range(n):
        if grades[i] == 2:
            print(names[i], "— приглашается на дополнительное занятие")
else:
    print("Данных нет")
```
`', '.join(...)` — объединяет строки списка через запятую (подробно проходится на 44-м уроке); вместо этого можно просто вывести список через `print`.
