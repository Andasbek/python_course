kaz_eng = {
    "сәлем": "hello", "мектеп": "school", "кітап": "book", "дос": "friend",

**Проект 2 «Система голосования» (полная версия)**
```python
names = input("Кандидаты (через запятую): ").split(",")
votes = {}
for n in names:
    votes[n.strip()] = 0
voted = set()

while True:
    voter = input("Имя избирателя (стоп — завершить): ").strip()
    if voter.lower() == "стоп":
        break
    if voter in voted:
        print("Вы уже проголосовали!")
        continue
    choice = input("Кандидат: ").strip()
    if choice not in votes:
        print("Такого кандидата нет")
        continue
    votes[choice] += 1
    voted.add(voter)
    print("Ваш голос принят")

total = sum(votes.values())
print("===== РЕЗУЛЬТАТЫ =====")
pairs = []
for name, v in votes.items():
    pairs.append((v, name))
pairs.sort(reverse=True)
for v, name in pairs:
    percent = 0
    if total > 0:
        percent = v / total * 100
    print(f"{name:<10}{v:>3} {percent:5.1f}% " + "█" * v)
print("Всего голосов:", total)

if total > 0:
    max_v = pairs[0][0]
    winners = []
    for v, name in pairs:
        if v == max_v:
            winners.append(name)
    if len(winners) == 1:
        print("Победитель:", winners[0])
    else:
        print("Равенство голосов:", ", ".join(winners), "— нужен второй тур")
```

**Задание 4 (структура)**
```python
candidates = {
    "Ученик года": ["Айгерим", "Бекзат"],
    "Спортсмен года": ["Дана", "Ерлан"],
}
votes = {}
voted = {}
for nom, people in candidates.items():
    votes[nom] = {}
    voted[nom] = set()
    for p in people:
        votes[nom][p] = 0

while True:
    voter = input("Избиратель (стоп): ").strip()
    if voter.lower() == "стоп":
        break
    for nom in candidates:
        if voter in voted[nom]:
            print(nom, "— голос уже отдан")
            continue
        choice = input(f"{nom} {candidates[nom]}: ").strip()
        if choice in votes[nom]:
            votes[nom][choice] += 1
            voted[nom].add(voter)
for nom, result in votes.items():
    print(nom, result)
```
