**Демо-проект: «Викторина о Казахстане» – состояние на конец урока 66 (MVP):**

```python
import random

QUESTIONS = {
    "Какой город – столица Казахстана?": "астана",
    "Национальная валюта Казахстана?": "тенге",
    "В каком году Казахстан провозгласил независимость?": "1991",
    "Одно из крупнейших озёр Казахстана – Балхаш или Байкал?": "балхаш",
    "Высокогорный каток в Алматы?": "медеу",
}


def normalize(text):
    """Готовит ответ к сравнению."""
    return text.strip().lower()


def check_answer(user_answer, correct_answer):
    """Возвращает True, если ответ верный."""
    return normalize(user_answer) == normalize(correct_answer)


def ask_question(number, question, answer):
    """Задаёт один вопрос и возвращает результат."""
    user = input(f"{number}. {question} ")
    if check_answer(user, answer):
        print("Верно!")
        return True
    print("Неверно. Правильный ответ:", answer)
    return False


def grade(score, total):
    """Переводит баллы в оценку."""
    percent = score / total * 100
    if percent >= 85:
        return "5 – отлично!"
    elif percent >= 65:
        return "4 – хорошо"
    elif percent >= 40:
        return "3 – неплохо"
    return "2 – готовься ещё"


def play(questions):
    """Игровой цикл, возвращает баллы."""
    items = list(questions.items())
    random.shuffle(items)
    score = 0
    number = 1
    for question, answer in items:
        if ask_question(number, question, answer):
            score += 1
        number += 1
    return score


score = play(QUESTIONS)
print(f"Результат: {score}/{len(QUESTIONS)}. Оценка: {grade(score, len(QUESTIONS))}")
```

```python
def check_answer(user_answer, correct_answer):
    return user_answer.strip().lower() == correct_answer.strip().lower()


def grade(score, total):
    percent = score / total * 100
    if percent >= 85:
        return "5"
    elif percent >= 65:
        return "4"
    elif percent >= 40:
        return "3"
    return "2"


def test_logic():
    tests = [
        (check_answer(" Астана", "астана"), True),
        (check_answer("Алматы", "астана"), False),
        (grade(10, 10), "5"),
        (grade(0, 10), "2"),
        (grade(4, 10), "3"),
    ]
    ok = True
    for i in range(len(tests)):
        got, expected = tests[i]
        if got != expected:
            print(f"ОШИБКА: тест {i + 1}, ожидалось {expected}, получено {got}")
            ok = False
    if ok:
        print("Все тесты пройдены!")


test_logic()
```
